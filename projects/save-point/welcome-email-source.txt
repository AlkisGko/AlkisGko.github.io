    1. AMPscript personalisation
    Look up the first name by recipient email. Use a friendly fallback
    for missing email addresses, missing records or blank first names.
  -->
  %%[
    VAR @email, @firstName, @name

    SET @name = "Adventurer"
    SET @email = AttributeValue("EmailAddress")

    IF NOT Empty(@email) THEN
      SET @firstName = Lookup(
        "Customers",
        "First_Name",
        "Email",
        @email
      )

      IF NOT Empty(@firstName) THEN
        SET @firstName = Trim(@firstName)

        IF NOT Empty(@firstName) THEN
          SET @name = ProperCase(@firstName)
        ENDIF
      ENDIF
    ENDIF
  ]%%
